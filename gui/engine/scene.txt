import pyglet
from pyglet import gl
from math import sqrt

from gui.engine.camera import Camera
from gui.engine.sphere_mesh import SphereMesh
from gui.engine.overlay import Overlay
from gui.engine.texture_manager import TextureManager
from services.calculation import RayCaster
from utils.math_utils import xyz_to_lonlat
from config.render_config import SPHERE_RADIUS

import logging
LOG = logging.getLogger(__name__)

class SceneWindow(pyglet.window.Window):
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        
        # Initialize Managers
        self.camera = Camera()
        self.overlay = Overlay()
        # Note: We don't need to pass the path anymore, the manager calculates it automatically
        self.tex_manager = TextureManager() 
        
        # Load Earth Texture (For 3D Sphere)
        earth_tex = self.tex_manager.load_texture("earth.jpg")
        self.sphere = SphereMesh(texture=earth_tex)
        
        # Load Background (For 2D Sprite)
        self.bg_sprite = self._setup_background()

    def _setup_background(self):
        # USE THE MANAGER: This ensures the path is correct
        bg_img = self.tex_manager.load_image("stars.jpg")
        
        if bg_img:
            # Create sprite using the loaded image
            sprite = pyglet.sprite.Sprite(bg_img, x=0, y=0)
            return sprite
        else:
            LOG.warning("Background image could not be loaded.")
            return None

    def on_resize(self, width, height):
        gl.glViewport(0, 0, width, height)
        return pyglet.event.EVENT_HANDLED

    def on_draw(self):
        self.clear()
        gl.glClear(gl.GL_COLOR_BUFFER_BIT | gl.GL_DEPTH_BUFFER_BIT)
        
        self.draw_background()
        self.setup_3d()
        
        # Apply Camera
        self.camera.apply_transform()
        
        # Draw Objects
        self.sphere.draw()
        self.overlay.draw()

    def draw_background(self):
        if not self.bg_sprite: return
        
        gl.glMatrixMode(gl.GL_PROJECTION)
        gl.glPushMatrix()
        gl.glLoadIdentity()
        gl.gluOrtho2D(0, self.width, 0, self.height)
        
        gl.glMatrixMode(gl.GL_MODELVIEW)
        gl.glPushMatrix()
        gl.glLoadIdentity()
        gl.glDisable(gl.GL_DEPTH_TEST)
        
        # Scale bg to fill window
        self.bg_sprite.scale_x = self.width / self.bg_sprite.width
        self.bg_sprite.scale_y = self.height / self.bg_sprite.height
        self.bg_sprite.draw()
        
        gl.glEnable(gl.GL_DEPTH_TEST)
        gl.glPopMatrix()
        gl.glMatrixMode(gl.GL_PROJECTION)
        gl.glPopMatrix()
        gl.glMatrixMode(gl.GL_MODELVIEW)

    def setup_3d(self):
        gl.glEnable(gl.GL_DEPTH_TEST)
        gl.glEnable(gl.GL_CULL_FACE)
        
        gl.glMatrixMode(gl.GL_PROJECTION)
        gl.glLoadIdentity()
        aspect = self.width / self.height
        gl.gluPerspective(60, aspect, 0.1, 100)
        
        gl.glMatrixMode(gl.GL_MODELVIEW)

    def on_mouse_scroll(self, x, y, scroll_x, scroll_y):
        self.camera.zoom(scroll_y)

    def on_mouse_drag(self, x, y, dx, dy, buttons, modifiers):
        if buttons == pyglet.window.mouse.LEFT:
            self.camera.rotate(dx, dy)
        elif buttons == pyglet.window.mouse.RIGHT:
            self.camera.pan(dx, dy)

    def on_key_press(self, symbol, modifiers):
        if symbol == pyglet.window.key.SPACE:
            self.camera.reset()
            self.overlay.clear()

    def on_mouse_press(self, x, y, button, modifiers):
        if button == pyglet.window.mouse.MIDDLE: 
            self.handle_click_ray(x, y)

    def handle_click_ray(self, x, y):
        try:
            model_matrix = (gl.GLdouble * 16)()
            proj_matrix = (gl.GLdouble * 16)()
            viewport = (gl.GLint * 4)()
            
            gl.glGetDoublev(gl.GL_MODELVIEW_MATRIX, model_matrix)
            gl.glGetDoublev(gl.GL_PROJECTION_MATRIX, proj_matrix)
            gl.glGetIntegerv(gl.GL_VIEWPORT, viewport)

            near_x, near_y, near_z = gl.GLdouble(), gl.GLdouble(), gl.GLdouble()
            far_x, far_y, far_z = gl.GLdouble(), gl.GLdouble(), gl.GLdouble()

            gl.gluUnProject(x, y, 0.0, model_matrix, proj_matrix, viewport, near_x, near_y, near_z)
            gl.gluUnProject(x, y, 1.0, model_matrix, proj_matrix, viewport, far_x, far_y, far_z)

            ray_origin = [near_x.value, near_y.value, near_z.value]
            ray_vec = [far_x.value - near_x.value, far_y.value - near_y.value, far_z.value - near_z.value]
            
            length = sqrt(sum(v*v for v in ray_vec))
            if length == 0: return
            ray_dir = [v/length for v in ray_vec]

            intersection = RayCaster.ray_sphere_intersection(ray_origin, ray_dir, [0,0,0], SPHERE_RADIUS)
            
            if intersection:
                self.overlay.add_pin(intersection)
                lon, lat = xyz_to_lonlat(*intersection)
                LOG.info(f"Clicked Lat: {lat:.2f}, Lon: {lon:.2f}")
                
        except Exception as e:
            LOG.error(f"Raycast error: {e}")
